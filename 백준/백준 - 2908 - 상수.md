백준 - 2908 - 상수
https://www.acmicpc.net/problem/2908


1. 두 수를 입력 받는다.
2. 두 수를 하나씩 반대로 배치한다.
3. 두 수를 비교하여 큰 수를 출력한다.

```java
import java.io.*;  
import java.util.*;  
  
public class Main {  
    public static void main(String[] args) throws IOException {  
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));  
        BufferedWriter bw = new BufferedWriter(new OutputStreamWriter(System.out));  
        StringBuilder sb = new StringBuilder();  
  
        // 두 수를 입력 받는다.  
        StringTokenizer st = new StringTokenizer(br.readLine());  
  
        int[] AandB = new int[2];  
  
        // 두 수를 하나씩 반대로 배치한다.  
        //변환 작업  
        for (int i = 0; i < 2; i++) {  
            String targetSt = st.nextToken();  
  
            char[] list = new char[targetSt.length()];  
            //순서 반대로 배치  
            for(int j = 0; j< targetSt.length(); j++){  
                list[(targetSt.length()-(j+1))] = targetSt.charAt(j);  
            }  
            //정수로 변환  
            int targetInt = 0;  
            int weight = 100;  
            for (int j = 0; j < 3; j++) {  
                targetInt += Integer.parseInt(String.valueOf(list[j])) * weight;  
                weight /= 10;  
            }  
            AandB[i] =  targetInt;  
        }  
        // 두 수를 비교하여 큰 수를 출력한다.  
        //출력  
        sb.append(Math.max(AandB[0], AandB[1]));  
        bw.write(sb.toString());  
  
        bw.flush();  
        bw.close();  
        br.close();  
    }  
  
}
```
