백준 - 1259 - 팰린드롬수

https://www.acmicpc.net/problem/1259


10988번 문제와 로직이 동일하지만, 입력 방식이나 출력 형태가 달랐기 떄문에 그 부분을 수정하고 로직을 함수에 넣는 방법으로 구현해서 성공하였다.

```java
import java.io.*;  
  
public class Main {  
    public static void main(String[] args) throws IOException {  
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));  
        BufferedWriter bw = new BufferedWriter(new OutputStreamWriter(System.out));  
        StringBuilder sb = new StringBuilder();  
  
        int word;  
        while ((word = Integer.parseInt(br.readLine())) != 0) { // 입력된 문자가 0이 아니면 계속 반복  
            String answer = isthisPalindrome(String.valueOf(word));  
            sb.append(answer).append('\n');  
        }  
  
        //출력  
        bw.write(sb.toString());  
        bw.flush();  
        bw.close();  
        br.close();  
    }  
  
    private static String isthisPalindrome(String word) {  
        //검사 횟수 설정  
        int count = word.length()/2;  
  
        //검사  
        int result = 1;  
        for (int i=0; i<count; i++) {  
            if (!(word.charAt(i) == word.charAt((word.length()-1)-i))) {  
                result = 0;  
                break;  
            }  
        }  
  
        String answer = result == 1 ? "yes" : "no"; // 소문자로 출력!!  
        return answer;  
    }  
  
}
```
