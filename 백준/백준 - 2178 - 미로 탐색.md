백준 - 2178 - 미로 탐색

https://www.acmicpc.net/problem/2178

bfs 알고리즘이란 걸 알아야 풀 수 있는 문제이다.

우선 GPT에게 간단한 미로 탐색 BFS 알고리즘을 짜게 한 후 그 코드를 바탕으로 BFS의 원리를 이해했다.
이해한 내용은 따로 메모해두었다.

그 후 문제에서 요구하는 입력 형식에 맞춰 코드를 수정하여 제출했다.

```java
import java.io.*;  
import java.util.*;  
  
public class Main {  
    static int[][] maze;  
    static boolean[][] visited;  
    static int[][] dist; //distance : 거리를 알기 위해 필요한 2차원 배열  
    static int n, m;  
  
    static int[] dx = {1, -1, 0, 0};  
    static int[] dy = {0, 0, 1, -1};  
  
    public static void main(String[] args) throws IOException {  
        BufferedWriter bw = new BufferedWriter(new OutputStreamWriter(System.out));  
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));  
        StringBuilder sb = new StringBuilder();  
  
        StringTokenizer st = new StringTokenizer(br.readLine());  
        int N = Integer.parseInt(st.nextToken());  
        int M = Integer.parseInt(st.nextToken());  
        n = N;  
        m = M;  
  
        int[][] Maze = new int[N][M];  
        visited = new boolean[N][M];  
        dist = new int[N][M];  
  
  
        for (int i = 0; i < N; i++) {  
            String mazeLine = br.readLine();  
            for (int j = 0; j < M; j++) {  
                Maze[i][j] = mazeLine.charAt(j) - '0';  
            }  
        }  
        maze = Maze;  
  
  
        bfs(0, 0);// x : 0, y : 0은 탐색의 시작점.  
  
        System.out.println(dist[N-1][M-1]); // 도착까지 최단 거리  
    }  
  
    static void bfs(int x, int y) {  
        Queue<int[]> queue = new LinkedList<>();  
  
        queue.add(new int[]{x, y});  
        visited[x][y] = true;  
        dist[x][y] = 1;  
  
        while (!queue.isEmpty()) {  
            int[] current = queue.poll();  
            int cx = current[0];  
            int cy = current[1];  
  
            for (int i = 0; i < 4; i++) {  
                int nx = cx + dx[i];  
                int ny = cy + dy[i];  
  
                // 범위 체크  
                if (nx < 0 || ny < 0 || nx >= n || ny >= m) continue;  
  
                // 벽이면 못 감  
                if (maze[nx][ny] == 0) continue;  
  
                // 이미 방문했으면 패스  
                if (visited[nx][ny]) continue;  
  
                visited[nx][ny] = true;  
                dist[nx][ny] = dist[cx][cy] + 1;  
                queue.add(new int[]{nx, ny});  
            }  
        }  
    }  
}
```
